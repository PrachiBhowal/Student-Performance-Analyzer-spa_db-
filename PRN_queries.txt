SELECT
    CASE
        WHEN marks >= 90 THEN 'A'
        WHEN marks >= 75 THEN 'B'
        WHEN marks >= 60 THEN 'C'
        WHEN marks >= 40 THEN 'D'
        ELSE 'F'
    END AS grade,
    COUNT(*) AS count
FROM students
GROUP BY grade
ORDER BY grade;

CREATE OR REPLACE VIEW vw_faculty_subjects AS
SELECT
    u.username AS faculty_username,
    s.subject_name,
    s.department,
    s.credits
FROM faculty_subjects fs
JOIN users u ON fs.faculty_user_id = u.id
JOIN subjects s ON fs.subject_id = s.subject_id;

CREATE OR REPLACE VIEW vw_student_subject_marks AS
SELECT
    st.name AS student_name,
    st.department,
    sub.subject_name,
    u.username AS faculty_username,
    ssm.exam_type,
    ssm.marks,
    ssm.updated_at
FROM student_subject_marks ssm
JOIN students st ON ssm.student_id = st.id
JOIN subjects sub ON ssm.subject_id = sub.subject_id
JOIN users u ON ssm.faculty_user_id = u.id;

ALTER TABLE students
ADD COLUMN total DECIMAL(6,2) DEFAULT 0,
ADD COLUMN average DECIMAL(5,2) DEFAULT 0;

UPDATE students s
JOIN (
    SELECT student_id,
           SUM(marks) AS total,
           AVG(marks) AS average
    FROM student_subject_marks
    GROUP BY student_id
) x ON s.id = x.student_id
SET s.total = x.total,
    s.average = x.average;

INSERT INTO student_subject_marks (student_id, subject_id, faculty_user_id, exam_type, marks)
VALUES (?, ?, ?, ?, ?)
ON DUPLICATE KEY UPDATE
marks = VALUES(marks),
faculty_user_id = VALUES(faculty_user_id),
updated_at = CURRENT_TIMESTAMP;
