Show all students with their PRN/username, sorted by student ID.
SELECT s.id, u.username AS prn, s.name, s.department, s.marks, s.enrolled_at
FROM students s
LEFT JOIN users u ON s.user_id = u.id
ORDER BY s.id;

Show dashboard statistics for total students, average marks, highest marks, and lowest marks.
SELECT COUNT(*), AVG(marks), MAX(marks), MIN(marks)
FROM students;

Show the top 3 performers by marks.
SELECT id, name, department, marks
FROM students
ORDER BY marks DESC
LIMIT 3;

Show marks distribution by grade bands.
SELECT
    CASE
        WHEN marks >= 90 THEN 'A'
        WHEN marks >= 75 THEN 'B'
        WHEN marks >= 60 THEN 'C'
        WHEN marks >= 40 THEN 'D'
        ELSE 'F'
    END AS grade,
    COUNT(*) AS count
FROM students
GROUP BY grade
ORDER BY grade;

Show department-wise average, highest, and lowest marks.
SELECT department, AVG(marks), MAX(marks), MIN(marks)
FROM students
GROUP BY department
ORDER BY department;

Show departments ranked by average marks.
SELECT department, AVG(marks) AS average_marks
FROM students
GROUP BY department
ORDER BY average_marks DESC;

Export students scoring above the overall average.
SELECT id, name, department, marks
FROM students
WHERE marks > (SELECT AVG(marks) FROM students)
ORDER BY marks DESC;

Fetch faculty-subject mapping through the faculty subjects view.
SELECT faculty_username, subject_name, department, credits
FROM vw_faculty_subjects
ORDER BY faculty_username, subject_name;

Fetch all student subject marks through the student subject marks view.
SELECT student_name, department, subject_name, faculty_username, exam_type, marks, updated_at

Fetch subject marks for the logged-in student.
SELECT sub.subject_name, u.username, ssm.exam_type, ssm.marks, ssm.updated_at
FROM student_subject_marks ssm
JOIN students st ON ssm.student_id = st.id
JOIN subjects sub ON ssm.subject_id = sub.subject_id
JOIN users u ON ssm.faculty_user_id = u.id
WHERE st.user_id = %s
ORDER BY sub.subject_name;

Create the faculty-subjects view used by the application.
CREATE OR REPLACE VIEW vw_faculty_subjects AS
SELECT
    u.username AS faculty_username,
    s.subject_name,
    s.department,
    s.credits
FROM faculty_subjects fs
JOIN users u ON fs.faculty_user_id = u.id
JOIN subjects s ON fs.subject_id = s.subject_id;

Create the student-subject-marks view used by the application.
CREATE OR REPLACE VIEW vw_student_subject_marks AS
SELECT
    st.name AS student_name,
    st.department,
    sub.subject_name,
    u.username AS faculty_username,
    ssm.exam_type,
    ssm.marks,
    ssm.updated_at
FROM student_subject_marks ssm
JOIN students st ON ssm.student_id = st.id
JOIN subjects sub ON ssm.subject_id = sub.subject_id
JOIN users u ON ssm.faculty_user_id = u.id;
